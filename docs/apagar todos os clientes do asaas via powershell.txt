$deleted = 0; $failed = 0; $offset = 0
do {
  $resp = Invoke-RestMethod -Uri "https://api-sandbox.asaas.com/v3/customers?limit=100&offset=$offset" -Headers $headers -Method Get
  foreach ($c in $resp.data) {
    try {
      Invoke-RestMethod -Uri "https://api-sandbox.asaas.com/v3/customers/$($c.id)" -Headers $headers -Method Delete | Out-Null
      Write-Host "OK $($c.id) $($c.name)"
      $deleted++
    } catch {
      Write-Host "FAIL $($c.id) $($c.name): $($_.Exception.Message)"
      $failed++
    }
  }
  $offset += 100
} while ($resp.hasMore -eq $true)

Write-Host "--- Done: deleted=$deleted failed=$failed ---"
$remaining = Invoke-RestMethod -Uri "https://api-sandbox.asaas.com/v3/customers?limit=1" -Headers $headers -Method Get
